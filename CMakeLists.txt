set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/compatibility.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/compatibility.json" compatibility)
string(JSON profile_count LENGTH "${compatibility}" profiles)
math(EXPR profile_last "${profile_count}-1")
set(supported_hashes "")
foreach(i RANGE ${profile_last})
  string(JSON module GET "${compatibility}" profiles ${i} module)
  if(module STREQUAL "assetbrowser")
    string(JSON hash GET "${compatibility}" profiles ${i} sha256)
    string(APPEND supported_hashes "\"${hash}\",\n")
  endif()
endforeach()
string(JSON logging_count LENGTH "${compatibility}" logging_profiles)
math(EXPR logging_last "${logging_count}-1")
set(logging_hashes "")
foreach(i RANGE ${logging_last})
  string(JSON hash GET "${compatibility}" logging_profiles ${i} sha256)
  string(APPEND logging_hashes "\"${hash}\",\n")
endforeach()
string(JSON steam_count LENGTH "${compatibility}" steam_profiles)
math(EXPR steam_last "${steam_count}-1")
set(steam_hashes "")
foreach(i RANGE ${steam_last})
  string(JSON hash GET "${compatibility}" steam_profiles ${i} sha256)
  string(APPEND steam_hashes "\"${hash}\",\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/supported_tools.h" "#pragma once\nnamespace ha { inline const char* supported_assetbrowser_hashes[]{${supported_hashes}}; inline const char* supported_logging_hashes[]{${logging_hashes}}; inline const char* supported_steam_hashes[]{${steam_hashes}}; }\n")
include_directories("${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ha_runtime PUBLIC bcrypt)
add_executable(tools_launcher native/tools_launcher.cpp)
target_include_directories(tools_launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(tools_launcher PRIVATE owned_process ha_runtime bcrypt advapi32)

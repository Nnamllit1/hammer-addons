cmake_minimum_required(VERSION 3.24)
project(hammer_addons VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Hammer Addons requires Windows x64 and MSVC.")
endif()
enable_language(ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
add_library(ha_runtime STATIC native/runtime.cpp)
target_include_directories(ha_runtime PUBLIC sdk/include native)
add_library(hammer SHARED native/proxy.cpp native/forward.asm native/hammer.def)
target_link_libraries(hammer PRIVATE ha_runtime)
add_library(assetbrowser SHARED native/proxy.cpp native/forward.asm native/assetbrowser.def)
target_compile_definitions(assetbrowser PRIVATE HA_ASSET_BROWSER)
target_link_libraries(assetbrowser PRIVATE ha_runtime)
add_executable(ha_host native/host.cpp)
target_link_libraries(ha_host PRIVATE ha_runtime)
add_library(hello SHARED addons/hello/hello.cpp)
target_include_directories(hello PRIVATE sdk/include)
add_library(test_original SHARED native/tests/original.cpp native/tests/original.def)
set_target_properties(test_original PROPERTIES OUTPUT_NAME hammer_original)
add_executable(proxy_test native/tests/proxy_test.cpp)
add_library(bad_abi SHARED native/tests/bad_abi.cpp)
target_include_directories(bad_abi PRIVATE sdk/include)
add_library(reentrant SHARED native/tests/reentrant.cpp)
target_include_directories(reentrant PRIVATE sdk/include)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Core Gui Widgets)
add_library(hammer_addons_ui SHARED native/ui.cpp)
target_link_libraries(hammer_addons_ui PRIVATE Qt5::Widgets)
# Qt's shared build uses the dynamic CRT.
set_target_properties(hammer_addons_ui PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_executable(ui_test native/tests/ui_test.cpp)
target_include_directories(ui_test PRIVATE native)
target_link_libraries(ui_test PRIVATE hammer_addons_ui Qt5::Widgets)
set_target_properties(ui_test PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
install(TARGETS hammer_addons_ui RUNTIME DESTINATION hammer-addons)
install(TARGETS assetbrowser ha_host RUNTIME DESTINATION .)
install(TARGETS hello RUNTIME DESTINATION hammer-addons/addons/hello)
install(FILES addons/hello/addon.ini DESTINATION hammer-addons/addons/hello)
install(DIRECTORY sdk/include/ DESTINATION sdk/include)
install(FILES docs/ADDON_FORMAT.md DESTINATION sdk)
install(FILES README.md VALIDATION.md THIRD_PARTY.md compatibility.json DESTINATION .)
install(FILES scripts/loader.py scripts/new-addon.py DESTINATION scripts)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY addons/hello DESTINATION addons FILES_MATCHING PATTERN "*.cpp" PATTERN "*.ini")

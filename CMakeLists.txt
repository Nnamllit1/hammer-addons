target_link_libraries(ha_host PRIVATE ha_runtime)
add_library(hello SHARED addons/hello/hello.cpp)

target_include_directories(bad_abi PRIVATE sdk/include)
add_library(reentrant SHARED native/tests/reentrant.cpp)

# Portable tools launcher: no replacement of a Valve DLL and no global loader settings.
add_library(owned_process STATIC native/owned_process.cpp)
target_include_directories(owned_process PUBLIC native)
add_library(hammer_addons_runtime SHARED native/tools_runtime.cpp)
target_link_libraries(hammer_addons_runtime PRIVATE ha_runtime shell32)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/compatibility.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/compatibility.json" compatibility)
string(JSON profile_count LENGTH "${compatibility}" profiles)
math(EXPR profile_last "${profile_count}-1")
set(supported_hashes "")
foreach(i RANGE ${profile_last})
  string(JSON module GET "${compatibility}" profiles ${i} module)
  if(module STREQUAL "assetbrowser")
    string(JSON hash GET "${compatibility}" profiles ${i} sha256)
    string(APPEND supported_hashes "\"${hash}\",\n")
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/supported_tools.h" "#pragma once\nnamespace ha { inline const char* supported_assetbrowser_hashes[]{${supported_hashes}}; }\n")
add_executable(tools_launcher native/tools_launcher.cpp)
target_include_directories(tools_launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(tools_launcher PRIVATE owned_process ha_runtime bcrypt advapi32)
add_executable(tools_fixture native/tests/tools_fixture.cpp)
target_include_directories(tools_fixture PRIVATE native)
target_link_libraries(tools_fixture PRIVATE Qt5::Widgets)
set_target_properties(tools_fixture PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_executable(launch_test native/tests/launch_test.cpp)
target_link_libraries(launch_test PRIVATE owned_process)
install(TARGETS tools_launcher hammer_addons_runtime hammer_addons_ui RUNTIME DESTINATION portable)
install(TARGETS hello RUNTIME DESTINATION portable/addons/hello)
install(FILES addons/hello/addon.ini DESTINATION portable/addons/hello)
install(FILES "Launch Workshop Tools.cmd" DESTINATION portable)

install(FILES docs/LAUNCHER.md DESTINATION portable RENAME README.md)
